\section{The paper}

\begin{frame}{The Problem: When and Where to Trust a Model}
  \begin{itemize}
    \item Deep RL needs \textbf{millions of interactions}; on real systems (robots, drones) each one is slow,
          expensive or risky.
    \item \textbf{Model-based RL (MBRL)} learns a model of the environment and \textbf{imagines} extra experience
          with it.
    \item But the model is never perfect: training on its \textbf{wrong predictions} misleads the agent.
  \end{itemize}
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{MBPO: \emph{when} to trust the model}
        Imagined rollouts get longer during training, following a \textbf{hand-designed schedule}, and the model is
        trusted \textbf{equally everywhere}.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{MACURA: \emph{where} to trust the model}
        Accuracy is \textbf{local}: every imagined rollout \textbf{stops where the model itself becomes uncertain}.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \small\textbf{Contributions}: a monotonic-improvement guarantee on a set $\E$ of accurate states; a cheap uncertainty
  measure (GJS) to build $\E$; adaptive rollouts with essentially \textbf{one hyperparameter} $\xi$; better results than
  MBPO and M2AC on MuJoCo.
\end{frame}

\begin{frame}{Background: SAC and Probabilistic Ensembles}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Soft Actor-Critic (SAC)}: the learner inside every method
      \begin{itemize}\small
        \item Off-policy actor-critic: learns from a \textbf{replay buffer}.
        \item \textbf{Maximum entropy}: reward plus randomness, so it keeps exploring:
              \[ J(\pi)=\textstyle\sum_t \mathbb{E}\big[r(s_t,a_t)+\alpha\,\mathcal{H}(\pi(\cdot\mid s_t))\big] \]
        \item Twin critics $Q_{\theta_1},Q_{\theta_2}$ (take the smaller), a Gaussian actor $\pi_\phi$, automatic
              temperature $\alpha$.
        \item One real step $\Rightarrow$ $G$ gradient updates (the \textbf{update-to-data ratio}).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Probabilistic ensemble (PE)}: the learned model
      \begin{itemize}\small
        \item $E=7$ neural networks, each trained on its own bootstrap sample of the real data.
        \item Each predicts a \textbf{Gaussian}: $\ptil_{\theta_e}(s'\mid s,a)=\mathcal{N}(\mu_{\theta_e},\Sigma_{\theta_e})$.
        \item \textbf{Aleatoric} uncertainty (noise of the world) $\Rightarrow$ wide $\Sigma_{\theta_e}$.
        \item \textbf{Epistemic} uncertainty (too little data) $\Rightarrow$ the members \textbf{disagree}.
        \item In a rollout, each step uses a \textbf{random member}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Dyna-Style MBRL: Imagined Rollouts}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth]{figures/paper/fig1_dyna.png}
        \caption{Dyna-style MBRL (paper, Fig.~1).}
      \end{figure}
    \end{column}
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}
        \item The agent acts in the real world $\Rightarrow$ real buffer $\Denv$; the model $\ptil$ learns from $\Denv$
              (supervised learning).
        \item \textbf{Branched rollouts}: start from real states $s_0\in\Denv$ and imagine a few steps with $\pi$ and
              $\ptil$ $\Rightarrow$ model buffer $\Dmod$.
        \item SAC trains mostly on $\Dmod$ (95\% imagined + 5\% real), with many updates per real step.
      \end{itemize}
      \textbf{Why it helps}: much more training data per real step, and it follows the \textbf{current} policy.\\[0.3em]
      \textbf{The risk}: longer rollouts drift away from the real data and errors add up; the agent learns to exploit
      the model's mistakes (\textbf{model exploitation}).\\[0.3em]
      $\Rightarrow$ The methods differ in \textbf{how many imagined steps they keep, and which}.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{MBPO and M2AC: Two Earlier Answers}
  \begin{columns}[T]
    \begin{column}{0.38\textwidth}
      \textbf{MBPO} (Janner et al., 2019)
      \begin{itemize}\small
        \item Every rollout has the \textbf{same length} $k$, which grows during training by a \textbf{schedule}
              chosen by hand per task (Humanoid $1\to25$, Walker 1).
        \item \textbf{Every} imagined step is used: no check of the model.
        \item A \textbf{fixed} number $G$ of SAC updates per real step.
      \end{itemize}
      $\Rightarrow$ answers \emph{when}, not \emph{where}.
    \end{column}
    \begin{column}{0.6\textwidth}
      \textbf{M2AC} (Pan et al., 2020): masking by uncertainty
      \begin{itemize}\small
        \item \textbf{One-vs-rest (OvR)} uncertainty: KL divergence between the member that made the prediction and a
              Gaussian fitted to the other members.
        \item All rollouts run the full horizon $H$; at step $h$ only the share $\frac{H-h}{2(H+1)}$ with the
              \textbf{lowest} uncertainty is kept ($\approx 25\%$ overall for $H=10$).
        \item \textbf{Rank-based}: a fixed share is kept even when all data are bad, or thrown away when all are good.
        \item \textbf{Reward penalty}: stored reward $r-\alpha\,u$ pushes the agent away from uncertain regions, which
              can also slow exploration.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Where to Trust Your Model?}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth]{figures/paper/fig2_where.png}
        \caption{The real data define a set $\E$ where the model is accurate (paper, Fig.~2).}
      \end{figure}
    \end{column}
    \begin{column}{0.58\textwidth}
      \small
      \begin{itemize}
        \item Inside $\E$ longer rollouts are useful; once a rollout \textbf{leaves} $\E$ its data are harmful.
        \item Some rollouts stay in $\E$ for long, others leave it at once: \textbf{a fixed length cannot fit both}.
      \end{itemize}
      \textbf{Theorem 4.1} (simplified): rollouts cut when they leave $\E$ give
      \[ \eta[\pi]\;\ge\;\tilde\eta[\pi]-C,\qquad
         C=2r_{\max}\textstyle\sum_{t}\gamma^t\sum_{\tau\le t}\Delta p_{\E}[\pi], \]
      $\Delta p_{\E}$ = the largest model error \textbf{inside} $\E$. If the policy improves in the model by more than
      $C$, it improves in the real world.
      \vspace{0.3em}

      \textbf{In practice}: the true error is unknown, so $\E$ is built from the model's own uncertainty:
      \[ \E=\{\,s\mid u(s,a)<\kappa,\ a\sim\pi(\cdot\mid s)\,\}. \]
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{GJS Uncertainty: Do the Members Agree?}
  \small
  Average disagreement over all \textbf{pairs} of members ($\mathcal{N}_e$ = member $e$'s Gaussian):
  \[
    \ugjs(s,a)=\frac{2}{E(E-1)}\sum_{e<f} D_{\mathrm{GJS}}(\mathcal{N}_e\|\mathcal{N}_f),\qquad
    D_{\mathrm{GJS}}=\tfrac12 D_{\mathrm{KL}}(\mathcal{N}_e\|\mathcal{N}_{ef})+\tfrac12 D_{\mathrm{KL}}(\mathcal{N}_f\|\mathcal{N}_{ef}),
  \]
  with $\mathcal{N}_{ef}$ the \textbf{geometric mean} of the two Gaussians (again a Gaussian).
  \begin{itemize}
    \item \textbf{Closed form} and symmetric: cheap enough for \textbf{every imagined step}.
    \item Large when two members are \textbf{confident but predict different things}.
  \end{itemize}
  \vspace{-0.3em}
  \begin{figure}
    \includegraphics[width=0.62\linewidth]{figures/paper/fig3_toy.png}
    \caption{Pendulum (paper, Fig.~3): (a) data, (b) true model error, (c) $\ugjs$, (d) the set $\E$.
             $\ugjs$ is low exactly where the real error is low.}
  \end{figure}
\end{frame}

\begin{frame}{Why a New Measure? GJS (MACURA) vs OvR (M2AC)}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth]{figures/paper/fig10_gjs_ovr.png}
        \caption{Top: GJS. Bottom: OvR. On the pendulum data (paper, Fig.~10).}
      \end{figure}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small
      \begin{itemize}
        \item \textbf{GJS} (b) is smooth and follows the data; its set $\E$ (c) has a clear border and matches the
              regions of \textbf{low real error} (d).
        \item \textbf{OvR} (f) is \textbf{noisy} and \textbf{over-confident}: low uncertainty in the centre, where the
              real error is high (e). Its set (g) does not match the accurate region (h).
        \item OvR looks at \textbf{one} member against the rest; GJS averages over \textbf{all pairs}, so one lucky
              member cannot hide the disagreement.
      \end{itemize}
      $\Rightarrow$ A reliable uncertainty is essential for deciding \textbf{where} to stop.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Uncertainty-Aware Rollouts}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \scriptsize
      \begin{algorithmic}
        \Require $\Denv,\Dmod,\ptil_{\theta_{1..E}},\pi,\zeta,\xi,T_{\max}$
        \State $s_0\sim\mathcal{U}(\Denv)$
        \For{$t=0,\dots,T_{\max}-1$}
          \State $e_t\sim\mathcal{U}(1..E)$,\ $a_t\sim\pi(\cdot\mid s_t)$
          \State $s_{t+1}\sim\ptil_{\theta_{e_t}}(\cdot\mid s_t,a_t)$,\ $r_{t+1}=r(s_t,a_t)$
          \State compute $\ugjs(s_t,a_t)$;\ \textbf{if} $t=0$: update $\kappa$
          \If{$\ugjs(s_t,a_t)<\kappa$}
            \State $\Dmod\gets\Dmod\cup\{(s_t,a_t,r_{t+1},s_{t+1})\}$
          \Else
            \State \textbf{break} \Comment{stop here, discard}
          \EndIf
        \EndFor
      \end{algorithmic}
      \vspace{0.2em}
      \centering\textcolor{navy}{Algorithm 2:} MACURA's rollout
    \end{column}
    \begin{column}{0.53\textwidth}
      \small
      \textbf{A self-tuning threshold $\kappa$}
      \begin{itemize}
        \item Each round, the $\zeta$-quantile ($\zeta=95\%$) of the \textbf{first-step} uncertainties
              $=$ ``what counts as certain now'';
              \ $\kappa=\frac{\xi}{K}\sum_{k=1}^{K}\hat u_{\mathrm{GJS},k}$.
        \item $T_{\max}=10$ and $\zeta$ are fixed; only $\xi$ is tuned per task.
      \end{itemize}
      \textbf{Updates follow the trusted data}
      \[ G=\Big\lfloor G_{\max}\,\tfrac{|\Dmod|}{|\Dmod|_{\max}}\Big\rceil \]
      Short rollouts (poor model) $\Rightarrow$ few updates; longer ones $\Rightarrow$ more.

      \textbf{Exploration} with \textbf{pink noise} (correlated in time) collects informative data and grows $\E$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Four Algorithms Side by Side}
  \centering
  \footnotesize
  \renewcommand{\arraystretch}{1.15}\setlength{\tabcolsep}{4pt}
  \begin{tabular}{>{\raggedright\arraybackslash\bfseries}p{2.2cm}*{3}{>{\raggedright\arraybackslash}p{2.35cm}}>{\raggedright\arraybackslash}p{3.2cm}}
    \toprule
    & \textbf{SAC} & \textbf{MBPO} & \textbf{M2AC} & \textbf{MACURA} \\
    \midrule
    world model & none & 7-network ensemble & same & same \\
    rollout length & -- & fixed $k$, schedule by hand & fixed horizon $H$ & \textbf{adaptive}: until $\ugjs\ge\kappa$ ($\le T_{\max}$) \\

    uncertainty & -- & none & OvR (one member vs rest) & GJS (all pairs) \\

    exploration (paper) & samples its policy & deterministic & deterministic & \textbf{pink noise} \\
    tuned per task & -- & the length schedule & masking, penalty $\alpha$ & only $\xi$ \\
    \bottomrule
  \end{tabular}
  \vspace{0.4em}

  \small All three model-based methods share the SAC learner, the ensemble and the 95\%/5\% data mix:
  they differ \textbf{only in which imagined data reach the learner}.
\end{frame}

\begin{frame}{How Long Are the Imagined Trips in Practice?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{figure}
        \includegraphics[width=0.92\linewidth]{figures/paper/fig17_rollout_len.png}
        \caption{Average rollout length on Walker (paper, Fig.~17).}
      \end{figure}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      \begin{itemize}
        \item \textbf{MBPO}: length 1 all training long (its tuned schedule for Walker).
        \item \textbf{M2AC}: a \textbf{constant} average, because a fixed share is always kept.
        \item \textbf{MACURA}: \textbf{short} while the model is poor (even below 1: it can drop the first step), then
              growing to about 7 as the model improves.
      \end{itemize}
      \vspace{0.3em}
      MACURA's updates follow the same curve: few at first, then more than MBPO's fixed number.
      The paper checks that \textbf{more updates alone} do not explain the gain: MBPO and M2AC with 40 updates per step
      stay behind.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Noise 1: Exploration Noise}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth]{figures/project/noise_types.png}
        \caption{Three kinds of noise added to the actions (same scale).}
      \end{figure}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Deterministic}: the policy's mean action, no noise (MBPO and M2AC in the paper).
        \item \textbf{SAC's own}: samples its Gaussian policy, like white noise with a learned width.
        \item \textbf{White noise}: independent every step; the pushes cancel out, so the agent barely leaves the
              states it already knows.
        \item \textbf{Pink noise} ($1/f$): \textbf{correlated in time}, it pushes in one direction for a while and
              reaches \textbf{new states}, without drifting away like a random walk (Eberhard et al., 2023).
      \end{itemize}
      \textbf{Why it matters for MACURA}: new real data at the border of $\E$ make the model accurate there
      $\Rightarrow$ $\E$ grows $\Rightarrow$ longer trusted rollouts.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Noise 2: Noise in the Environment}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth]{figures/paper/fig16_noise.png}
        \caption{HalfCheetah with hidden action noise $\sigma\in\{0.05,0.1,0.2\}$ (paper, Fig.~16).}
      \end{figure}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Process noise}: the world itself is random. The paper adds Gaussian noise to every action, and the
              agent does not see it.
        \item A model can learn noise only as a \textbf{spread}: each member predicts a wide Gaussian
              (\textbf{aleatoric}).
        \item GJS compares members: if all predict the \textbf{same} wide Gaussian, $\ugjs\approx0$ and the step is
              still trusted. MACURA stops for \textbf{disagreement} (epistemic), not for noise.
        \item MACURA stays ahead of MBPO and M2AC at all three noise levels; the gap shrinks as the noise grows.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Results on MuJoCo}
  \begin{figure}
    \includegraphics[width=0.7\linewidth]{figures/paper/fig4_mujoco.png}
    \caption{Return while learning on five MuJoCo tasks; mean of 5 seeds with 95\% CI (paper, Fig.~4).}
  \end{figure}
  \vspace{-0.3em}
  \begin{itemize}\small
    \item MACURA learns \textbf{faster} and ends \textbf{higher} than MBPO and M2AC, most clearly on Humanoid and Ant.
    \item It often reaches model-free SAC's final return with a fraction of the real data.
    \item Why: long rollouts where the model is reliable, none where it is not $\Rightarrow$ \textbf{more and cleaner} data.
  \end{itemize}
\end{frame}

\begin{frame}{Exploration Changes the Results}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth]{figures/paper/fig5_exploration.png}
        \caption{Humanoid: deterministic (D), white noise (WN) and pink noise (PN) exploration (paper, Fig.~5).}
      \end{figure}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Deterministic}: limited but \textbf{stable}, for both methods.
        \item \textbf{White noise}: can be very strong for MACURA, but it \textbf{destabilises} MBPO.
        \item \textbf{Pink noise}: the best balance of performance and stability.
        \item MACURA gains the most from better exploration: new data grow $\E$, exactly where it uses the model.
        \item With deterministic interaction for \textbf{all} methods, MACURA is still the strongest on most tasks
              (paper appendix).
      \end{itemize}
      \vspace{0.3em}
      \footnotesize Note: the main comparison gives pink noise only to MACURA; I kept this protocol in my project.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What Matters? Ablations and the Threshold}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{figure}
        \includegraphics[width=0.85\linewidth]{figures/paper/fig7_ablation.png}
        \caption{Humanoid, one building block changed (paper, Fig.~7).}
      \end{figure}
      \vspace{-0.8em}
      \begin{figure}
        \includegraphics[width=0.42\linewidth]{figures/paper/fig8_kappa.png}
        \caption{$\kappa$ during training (paper, Fig.~8).}
      \end{figure}
    \end{column}
    \begin{column}{0.51\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Rank-based} masking (M2AC's rule) instead of a threshold: much worse.
        \item \textbf{OvR} uncertainty instead of GJS: \textbf{diverges}, unreliable data get through.
        \item A \textbf{fixed} $\kappa$: unstable at some stage of training. The adaptive $\kappa$ starts high (rough data
              are enough at first) and then drops (precise data are needed later): a schedule that \textbf{emerges by itself}.
        \item A \textbf{fixed} number of updates: a moderate loss.
        \item $\xi$ works over a \textbf{wide range}; too large exploits model errors, too small overfits.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Paper in Short}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Takeaways}
      \begin{itemize}\small
        \item Change the question from \emph{when} to \emph{where} to trust the model.
        \item Rollouts inside a set $\E$ of accurate states $\Rightarrow$ monotonic improvement.
        \item GJS: a cheap, closed-form, reliable measure of disagreement (not of noise).
        \item Rollouts stop at the first uncertain step; $\kappa$ and the updates $G$ tune themselves.
        \item One knob, $\xi$; better data efficiency than MBPO and M2AC.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Limits}
      \begin{itemize}\small
        \item $\xi$ still differs a lot between tasks (0.3 to 30).
        \item Very long training destabilises MBPO and MACURA alike.
        \item The main comparison uses \textbf{different exploration} (pink noise vs deterministic).
        \item Only simulated locomotion tasks.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1em}
  \centering
  \colorbox{myblue!10}{\parbox{0.9\linewidth}{\centering\small
    \textbf{My question}: does MACURA's advantage hold on a \textbf{new task I built myself}, with the same compute
    for every method, and does it really trust its model less \textbf{where the model is wrong}?}}
\end{frame}

\begin{frame}{The Chute Trap: Where the Model Should Disagree}
  \begin{columns}[T]

        \item This is \textbf{rare} in the data and \textbf{hard to predict} $\Rightarrow$ the members should
              \textbf{disagree} there (epistemic).
        \item Exactly where MACURA should \textbf{stop imagining}, and where MBPO would keep learning from wrong
              predictions.
      \end{itemize}

\end{frame}

\begin{frame}{Limitations (1/2): Training Budget and Statistics}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Few training steps}
      \begin{itemize}\small
        \item Only \textbf{50,000 real steps} per run (5,000 of them random); the paper uses 125k--400k.
        \item The curves are \textbf{still rising} at the end: I compare \textbf{early learning}, not final performance.
        \item The task is \textbf{not solved}: 31\% of a lap per 10 s, against 54\% for a simple autopilot at
              1.5 m/s.
        \item Kaggle limits (12 h per session, about 30 GPU h per week and account) did not allow longer runs.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Few seeds}
      \begin{itemize}\small
        \item 4 seeds per algorithm in the final study (8 for MACURA vs MBPO).
        \item 7 of 8 is \textbf{consistent}, but only at the edge of statistical significance ($p\approx0.035$).
        \item The confidence intervals are wide; M2AC and MACURA overlap.
        \item \textbf{No hyperparameter search}: $\xi=0.5$, MBPO's length $1\to10$ and M2AC's penalty were set once,
              not tuned per method.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Limitations (2/2): Fairness and the Task}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Fairness}
      \begin{itemize}\small
        \item \textbf{Exploration differs}: only MACURA uses pink noise (the paper's protocol). Part of its gain may come
              from exploration; the paper shows pink noise helps MBPO too.
        \item MBPO and M2AC use the paper's rules as they are; the paper tunes MBPO's length schedule per task, I did
              not.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{The task}
      \begin{itemize}\small
        \item \textbf{One task}, designed and changed by me five times: a risk of picking a task that suits MACURA
              (mitigated by rules fixed in advance and new seeds).
        \item The trained drones \textbf{rarely reach the chute trap}, so the ``where'' mechanism was tested less than
              planned.
        \item The disagreement predicts the real model error only \textbf{weakly} here.
        \item \textbf{Simulation only}: nothing was flown on a real drone.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusion}

\begin{frame}{Lessons and Next Steps}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Lessons}
      \begin{itemize}\small
        \item The task decides everything: MACURA only helps if the model is uncertain \textbf{where it matters}.
        \item Fix the decision rules \textbf{before} the runs, and confirm on \textbf{new seeds}.
        \item Give every method the \textbf{same compute} (updates per step), or the comparison says nothing.
        \item Separate the two noises: exploration noise changes the data, environment noise the model.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Next steps}
      \begin{itemize}\small
        \item Train \textbf{longer} (100k--200k steps), so the drones fly the chute fast and the trap matters.
        \item \textbf{More seeds} (10 or more) for clear statistics.
        \item The \textbf{same exploration} for all methods (already possible in the code).
        \item Tune $\xi$, and compare with the paper's settings.
        \item Test the best policies on a \textbf{real drone}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item \textbf{The paper}: MACURA trusts its learned model only \textbf{where it trusts itself}. Rollouts stop at the
          first uncertain step (GJS $\ge\kappa$), $\kappa$ and the updates tune themselves, and one knob ($\xi$) is left.
          GJS reacts to disagreement, not to noise.
    \item \textbf{My project}: on a racing drone with wind, gusts, a package and a chute, with the same compute for all:
      \begin{itemize}
        \item MACURA had the \textbf{best test return} (333 vs 267 M2AC, 158 MBPO, $-312$ SAC);
        \item it beat MBPO in \textbf{7 of 8 seeds} and M2AC in 3 of 4, with no collapse;
        \item it trusted its model \textbf{less in fast descents} (67\%) than in normal flight (94\%).
      \end{itemize}
    \item With only 50k steps and few seeds this is \textbf{early-learning evidence}: consistent, but not yet proof.
    \item Model-based RL is far more data-efficient than model-free SAC here, and MACURA was the strongest of the three
          model-based methods on this task.
  \end{itemize}
\end{frame}
