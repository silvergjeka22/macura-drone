# ── macura-drone requirements ────────────────────────────────────────────────
# Targeted at Google Colab (Python 3.12, CUDA GPU). All deps below install
# cleanly on Colab's current runtime + numpy 2.x.
#
# Backbone note: we build on Stable-Baselines3 (maintained) for the SAC learner,
# NOT mbrl-lib. mbrl-lib is unmaintained and not installable on Python 3.12; the
# probabilistic ensemble and the MACURA / MBPO / M2AC rollout logic are our own
# pure-function modules (models/ensemble.py, algorithms/*.py).

# --- core scientific stack -------------------------------------------------
# DO NOT cap numpy below 2.0 on Colab: gymnasium/opencv/etc. are compiled
# against numpy 2.x, so pinning <2 breaks the ABI ("numpy.dtype size changed").
numpy>=1.26
scipy>=1.10
matplotlib>=3.7
pyyaml>=6.0
pandas>=2.0
tqdm>=4.65

# --- deep learning ---------------------------------------------------------
torch>=2.1            # Colab ships a compatible CUDA build

# --- simulation ------------------------------------------------------------
mujoco>=3.1           # MuJoCo physics (loads the Skydio X2 MJCF)
gymnasium>=0.29       # env API for our drone task
imageio>=2.31         # save rollout frames / videos for the study plots
imageio-ffmpeg>=0.4   # mp4 encoding

# --- RL backbone (SAC learner; shared by all four algorithms) --------------
stable-baselines3>=2.3   # maintained SAC implementation, Colab-compatible
